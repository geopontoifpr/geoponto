class FechamentoMensalModel {
  final String? id;

  final int minutosPrevistos;
  final int minutosTrabalhados;
  final int minutosAbonados;
  final int saldoMinutos;
  final bool fechada;
  final DateTime? criadoEm;
  final DateTime? atualizadoEm;

  FechamentoMensalModel({
    this.id,
    required this.usuarioId,
    required this.mesReferencia,
    required this.minutosPrevistos,
    required this.minutosTrabalhados,
    required this.minutosAbonados,
    required this.saldoMinutos,
    this.fechada = true,
    this.criadoEm,
    this.atualizadoEm,
  });

  factory FechamentoMensalModel.fromMap(Map<String, dynamic> map) {
    return FechamentoMensalModel(
      id: map['id']?.toString(),
      usuarioId: map['usuario_id'].toString(),
      mesReferencia: DateTime.parse(map['mes_referencia'].toString()),
      minutosPrevistos: map['minutos_previstos'] as int,
      minutosTrabalhados: map['minutos_trabalhados'] as int,
      minutosAbonados: map['minutos_abonados'] as int,
      saldoMinutos: map['saldo_minutos'] as int,
      fechada: map['fechada'] as bool? ?? false,
      criadoEm: map['criado_em'] != null ? DateTime.parse(map['criado_em'].toString()) : null,
      atualizadoEm: map['atualizado_em'] != null ? DateTime.parse(map['atualizado_em'].toString()) : null,
    );
  }

  Map<String, dynamic> toMap() {
    // Formato 'YYYY-MM-01' para a coluna date do PostgreSQL
    final mesRefStr = "${mesReferencia.year}-${mesReferencia.month.toString().padLeft(2, '0')}-01";
    return {
      if (id != null) 'id': id,
      'usuario_id': usuarioId,
      'mes_referencia': mesRefStr,
      'minutos_previstos': minutosPrevistos,
      'minutos_trabalhados': minutosTrabalhados,
      'minutos_abonados': minutosAbonados,
      'saldo_minutos': saldoMinutos,
      'fechada': fechada,
    };

enum StatusDiaCalendario { neutro, verde, laranja, vermelho }

class FrequenciaService {
  final FrequenciaRepository _frequenciaRepo;
  final OcorrenciaRepository _ocorrenciaRepo;

  FrequenciaService({
    FrequenciaRepository? frequenciaRepo,
    OcorrenciaRepository? ocorrenciaRepo,
  })  : _frequenciaRepo = frequenciaRepo ?? FrequenciaRepository(),
        _ocorrenciaRepo = ocorrenciaRepo ?? OcorrenciaRepository();

  Future<DateTime> obterDataCadastroUsuario(String usuarioId) {
    return _frequenciaRepo.obterDataCadastroUsuario(usuarioId);
  }

  Future<int> obterCargaHorariaSemanal(String usuarioId) {
    return _frequenciaRepo.obterCargaHorariaSemanal(usuarioId);
  }

  Future<Map<int, List<RegistroPontoModel>>> obterPontosAgrupadosPorDia({
    required String usuarioId,
    required int ano,
    required int mes,
  }) async {
    final pontos = await _frequenciaRepo.listarPontosPorMes(
      usuarioId: usuarioId,
      ano: ano,
      mes: mes,
    );
    final mapPontos = <int, List<RegistroPontoModel>>{};
    for (final p in pontos) {
      final dia = p.criadoEm.toLocal().day;
      mapPontos.putIfAbsent(dia, () => []).add(p);
    }
    return mapPontos;
  }

  Future<Map<int, List<OcorrenciaModel>>> obterOcorrenciasAgrupadasPorDia({
    required String usuarioId,
    required int ano,
    required int mes,
  }) async {
    final ocorrencias = await _ocorrenciaRepo.listarPorMes(
      usuarioId: usuarioId,
      ano: ano,
      mes: mes,
    );
    final mapOcorrencias = <int, List<OcorrenciaModel>>{};
    for (final oc in ocorrencias) {
      final dia = oc.dataInicio.day;
      mapOcorrencias.putIfAbsent(dia, () => []).add(oc);
    }
    return mapOcorrencias;
  }

  
  Duration calcularHorasDoDia({
      required List<RegistroPontoModel> batidasDoDia,
      required List<OcorrenciaModel> ocorrenciasDoDia,
    }) {
      final batidas = List<RegistroPontoModel>.from(batidasDoDia)
        ..sort((a, b) => a.criadoEm.compareTo(b.criadoEm));

      Duration total = Duration.zero;

      for (int i = 0; i + 1 < batidas.length; i += 2) {
        final diferenca =
            batidas[i + 1].criadoEm.difference(batidas[i].criadoEm);

        if (!diferenca.isNegative) {
          total += diferenca;
        }
      }

      return total;
  }


  StatusDiaCalendario avaliarStatusDia({
    required DateTime dataDia,
    required DateTime dataCadastroUsuario,
    required int cargaHorariaSemanal,
    required List<RegistroPontoModel> batidas,
    required List<OcorrenciaModel> ocorrencias,
  }) {
    final hoje = DateTime.now();
    final hojeSemHora = DateTime(hoje.year, hoje.month, hoje.day);
    final cadastroSemHora = DateTime(
      dataCadastroUsuario.year,
      dataCadastroUsuario.month,
      dataCadastroUsuario.day,
    );

    final isFimDeSemana =
        dataDia.weekday == DateTime.saturday || dataDia.weekday == DateTime.sunday;

    if (dataDia.isAfter(hojeSemHora)) return StatusDiaCalendario.neutro;
    if (dataDia.isBefore(cadastroSemHora) && batidas.isEmpty) {
      return StatusDiaCalendario.neutro;
    }
    if (isFimDeSemana && batidas.isEmpty) {
      return StatusDiaCalendario.neutro;
    }

    if (batidas.isEmpty) {
      final temAbonoAprovado = ocorrencias.any((o) => o.aprovada);
      if (temAbonoAprovado) return StatusDiaCalendario.verde;

      if (dataDia.isAtSameMomentAs(hojeSemHora)) {
        return StatusDiaCalendario.neutro;
      }
      return StatusDiaCalendario.vermelho;
    }

    if (batidas.length.isOdd) {
      return StatusDiaCalendario.laranja;
    }

    final metaMinutosDia = (cargaHorariaSemanal / 5.0 * 60.0).round();
    final trabalhadoMinutos = calcularHorasDoDia(
      batidasDoDia: batidas,
      ocorrenciasDoDia: ocorrencias,
    ).inMinutes;

    if (trabalhadoMinutos >= (metaMinutosDia - 15)) {
      return StatusDiaCalendario.verde;
    } else {
      return StatusDiaCalendario.laranja;
    }
  }

          .select('criado_em')

          .maybeSingle();

      }

  Future<int> obterCargaHorariaSemanal(String usuarioId) async {

          .select('jornada_id')
          .eq('usuario_id', usuarioId)
          .order('criado_em', ascending: false)
          .limit(1)
          .maybeSingle();

      if (vinculo != null && vinculo['jornada_id'] != null) {
        final jornada = await _client

            .select('carga_horaria_sem')

      final minutosDia = calcularHorasDoDia(d).inMinutes;
      totalTrabalhadoMinutos += minutosDia;
      metaMensalMinutos += obterMinutosPrevistosDoDia(d);
    }

    final faltamMinutos =
        (metaMensalMinutos - totalTrabalhadoMinutos)
            .clamp(0, metaMensalMinutos);

    final progresso = metaMensalMinutos > 0
        ? (totalTrabalhadoMinutos / metaMensalMinutos).clamp(0.0, 1.0)
        : 0.0;

    return (
      trabHoras: totalTrabalhadoMinutos ~/ 60,
      trabMin: totalTrabalhadoMinutos % 60,
      faltamHoras: faltamMinutos ~/ 60,
      faltamMinRest: faltamMinutos % 60,
      progresso: progresso,
    );

  // Atualize o retorno de calcularExtratoDetalhado() para calcular também os minutos abonados:
  ({
    int horasPrevistas,
    Duration totalTrabalhado,
    int saldoMinutos,
    int diasTrabalhados,
    int minutosCredito,
    int minutosDebito,
    int minutosAbonados,
  }) calcularExtratoDetalhado() {
    final hoje = DateTime.now();
    final hojeSemHora = DateTime(hoje.year, hoje.month, hoje.day);
    final metaDiariaMinutos = (cargaHorariaSemanal / 5.0 * 60.0).round();

    int diasUteisNoMes = 0;
    int diasTrabalhados = 0;
    int minutosCredito = 0;
    int minutosDebito = 0;
    int minutosAbonados = 0;
    Duration totalTrabalhado = Duration.zero;

    for (int d = 1; d <= totalDiasMes; d++) {
      final dt = DateTime(mesAtual.year, mesAtual.month, d);
      final isDiaUtil =
          dt.weekday != DateTime.saturday && dt.weekday != DateTime.sunday;

      if (isDiaUtil) {
        diasUteisNoMes++;
      }

      final duracaoDia = calcularHorasDoDia(d);
      final minDia = duracaoDia.inMinutes;
      totalTrabalhado += duracaoDia;

      // Calcula minutos abonados do dia
      final ocs = ocorrenciasPorDia[d] ?? [];
      for (final oc in ocs) {
        if (oc.aprovada && oc.minutosAbonados != null) {
          minutosAbonados += oc.minutosAbonados!;
        }
      }

      if (minDia > 0 || (pontosPorDia[d]?.isNotEmpty ?? false)) {
        diasTrabalhados++;
      }

      if (!dt.isAfter(hojeSemHora)) {
        if (isDiaUtil) {
          final diff = minDia - metaDiariaMinutos;
          if (diff > 0) {
            minutosCredito += diff;
          } else if (diff < 0 && minDia > 0) {
            minutosDebito += diff.abs();
          }
        } else if (minDia > 0) {
          minutosCredito += minDia;
        }
      }
    }

    final horasPrevistas = (diasUteisNoMes * (cargaHorariaSemanal / 5.0)).round();
    final saldoMinutos = minutosCredito - minutosDebito;

    return (
      horasPrevistas: horasPrevistas,
      totalTrabalhado: totalTrabalhado,
      saldoMinutos: saldoMinutos,
      diasTrabalhados: diasTrabalhados,
      minutosCredito: minutosCredito,
      minutosDebito: minutosDebito,
      minutosAbonados: minutosAbonados,
    );
  }

  
  int obterCargaHorariaDoDia(int dia) {
    final data = DateTime(mesAtual.year, mesAtual.month, dia);

        '${data.year}-${data.month.toString().padLeft(2, '0')}-${data.day.toString().padLeft(2, '0')}';

      final inicio = vinculo['data_inicio_vigencia']?.toString();
      final fim = vinculo['data_fim_vigencia']?.toString();

      if (inicio == null || inicio.compareTo(dataTexto) > 0) {
        continue;
      }

      if (fim != null && fim.compareTo(dataTexto) < 0) {
        continue;
      }

      final jornada = vinculo['jornadas_trabalho'];

      if (jornada is Map && jornada['carga_horaria_sem'] != null) {
        return (jornada['carga_horaria_sem'] as num).toInt();
      }
    }

    return cargaHorariaSemanal;
  }

  int obterMinutosPrevistosDoDia(int dia) {
    final data = DateTime(mesAtual.year, mesAtual.month, dia);

    if (data.weekday == DateTime.saturday ||
        data.weekday == DateTime.sunday) {
      return 0;
    }

    return (obterCargaHorariaDoDia(dia) / 5 * 60).round();
  }

        latitudeEmpresa = dados['latitude_empresa'] as double;
        longitudeEmpresa = dados['longitude_empresa'] as double;
        raioPermitido = dados['raio_permitido_m'] as double;
      }

      final posicao = await _locationService.obterLocalizacaoAtual();

      if (posicao.isMocked) {
        posicaoAtual = posicao;
        distanciaEmpresa = null;
        dentroDoRaio = false;
        avisoLocalizacao =
            'Localização simulada (Fake GPS) detectada. Não é permitido bater ponto.';
        erro = avisoLocalizacao;
        return false;
      }

      if (posicao.accuracy > 50) {
        posicaoAtual = posicao;
        distanciaEmpresa = null;
        dentroDoRaio = false;
        avisoLocalizacao =
            'Sinal de GPS impreciso (±${posicao.accuracy.toStringAsFixed(0)}m). Vá para um local aberto e tente novamente.';
        erro = avisoLocalizacao;
        return false;
      }

      posicaoAtual = posicao;
      final distancia = CalculadoraDistanciaUtil.calcularDistancia(
        latitudeOrigem: posicao.latitude,
        longitudeOrigem: posicao.longitude,
        latitudeDestino: latitudeEmpresa!,
        longitudeDestino: longitudeEmpresa!,
      );
      distanciaEmpresa = distancia;
      dentroDoRaio = distancia <= raioPermitido!;

      if (!dentroDoRaio) {
        avisoLocalizacao =
            'Você está fora do raio permitido da empresa (${distancia.toStringAsFixed(0)}m de distância; máximo permitido: ${raioPermitido!.toStringAsFixed(0)}m).';
        erro = avisoLocalizacao;
        return false;
      }

      return true;
    } catch (e) {
      posicaoAtual = null;
      distanciaEmpresa = null;
      dentroDoRaio = false;
      avisoLocalizacao = _mensagemErro(e);
      erro = avisoLocalizacao;
      return false;
    } finally {
      atualizandoLocalizacao = false;
      notifyListeners();
    }
  }

  Future<void> carregarRegistros() async {

  }

  Future<bool> baterPonto() async {
    if (registrando || atualizandoLocalizacao) return false;

    registrando = true;
    erro = null;
    notifyListeners();

    try {
      // 1. Executa a verificação de GPS e Raio ao clicar no botão
      final gpsValido = await atualizarLocalizacao();
      if (!gpsValido || !dentroDoRaio || !precisaoValida || posicaoAtual == null) {
        erro ??= 'Não foi possível validar sua localização dentro do raio da empresa.';
        return false;
      }

      // 2. Solicita a Biometria (apenas em dispositivos móveis; na Web pula biometria)
      if (!kIsWeb) {
        final autenticado = await _biometriaService.autenticar();
        if (!autenticado) {
          erro = 'Autenticação biométrica cancelada. O ponto não foi registrado.';
          return false;
        }
      }

      // 3. Registra o ponto no Supabase
      await _pontoService.registrarPonto(
        usuarioId: usuarioId,
        latitude: posicaoAtual!.latitude,
        longitude: posicaoAtual!.longitude,
      );
      await carregarRegistros();
      return true;
    } catch (e) {
      erro = _mensagemErro(e);
      return false;
    } finally {
      registrando = false;
      notifyListeners();
    }
  }

  String _mensagemErro(Object erro) {
    if (erro is LocalizacaoDesativadaException) {
      return 'A localização do aparelho está desativada. Ative o GPS e tente novamente.';
    }
    if (erro is PermissaoLocalizacaoNegadaException) {
      return 'A permissão de localização foi negada. Ela é necessária para bater ponto.';
    }
    if (erro is PermissaoLocalizacaoNegadaPermanentementeException) {
      return 'A permissão de localização está bloqueada nas configurações do aparelho.';
    }
    return erro.toString().replaceFirst('Exception: ', '');
  }

enum TipoRegistroEnum {
  entrada1('entrada_1', 'Entrada 1'),
  saida1('saida_1', 'Saída 1 (Almoço)'),
  entrada2('entrada_2', 'Entrada 2 (Retorno)'),
  saida2('saida_2', 'Saída 2 (Fim do Expediente)');

  final String valorBanco;
  final String titulo;

  const TipoRegistroEnum(this.valorBanco, this.titulo);

  // Fabrica o Enum a partir do que vem do banco de dados (trata entry1, entrada_1, etc.)
  factory TipoRegistroEnum.fromBanco(String valor) {
    final normalizado = valor.toLowerCase().replaceAll('_', '');
    switch (normalizado) {
      case 'entrada1':
      case 'entry1':
        return TipoRegistroEnum.entrada1;
      case 'saida1':
      case 'exit1':
      case 'saidaalmoco':
        return TipoRegistroEnum.saida1;
      case 'entrada2':
      case 'entry2':
      case 'retorno':
        return TipoRegistroEnum.entrada2;
      case 'saida2':
      case 'exit2':
        return TipoRegistroEnum.saida2;
      default:
        return TipoRegistroEnum.entrada1; // Fallback de segurança
    }
  }
}
